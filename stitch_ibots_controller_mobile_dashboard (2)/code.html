// Sample BLE characteristic packet reception:

void onPacketReceived(char cmd) {
  switch(cmd) {
    case 'U': moveForward(); break;
    case 'D': moveBackward(); break;
    case 'L': turnLeft(); break;
    case 'R': turnRight(); break;
    case 'J': handleAnalog(joyX, joyY); break; // Joystick
    case 'T': actionTriangle(); break; // Tri
    case 'S': actionSquare(); break;   // Sqr
    case 'C': actionCircle(); break;   // Cir
    case 'X': actionCross(); break;    // X
    case '1': enableMotors(); break;   // START
    case '0': haltMotors(); break;     // PAUSE
  }
}
